Here's a sliding window rate limiter using Redis. It tracks requests per user with automatic expiry:

rate-limiter.ts Copy
import { Redis } from '@upstash/redis' const redis = Redis.fromEnv() export async function rateLimit( userId: string, limit = 60, window = 60 ) { const key = `rate:${userId}` const now = Date.now() await redis.zremrangebyscore( key, 0, now - window * 1000 ) const count = await redis.zcard(key) if (count >= limit) { return { success: false, remaining: 0 } } await redis.zadd(key, { score: now, member: now }) await redis.expire(key, window) return { success: true, remaining: limit - count - 1 } }

This gives you 60 requests per 60-second sliding window per user. The sorted set stores each request timestamp, and old entries expire automatically.
